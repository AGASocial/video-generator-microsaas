    ```typescript
    /**
     * Kling AI JWT Authentication
     * Generates short-lived JWT tokens for Kling API requests.
     * Server-side only — never import from client components.
     *
     * JWT spec verified from: github.com/199-mcp/mcp-kling/blob/main/generate-jwt.mjs
     * Algorithm: HS256 (HMAC-SHA256)
     * Claims: iss=KLING_ACCESS_KEY, exp=now+1800, nbf=now-5
     */
    import crypto from "crypto";

    function base64url(input: string): string {
      return Buffer.from(input)
        .toString("base64")
        .replace(/\+/g, "-")
        .replace(/\//g, "_")
        .replace(/=/g, "");
    }

    export function generateKlingToken(): string {
      const accessKey = process.env.KLING_ACCESS_KEY;
      const secretKey = process.env.KLING_SECRET_KEY;

      if (!accessKey || !secretKey) {
        console.error("[Kling] KLING_ACCESS_KEY or KLING_SECRET_KEY is not configured");
        throw new Error("Kling credentials not configured");
      }

      const header = base64url(JSON.stringify({ alg: "HS256", typ: "JWT" }));
      const now = Math.floor(Date.now() / 1000);
      const payload = base64url(
        JSON.stringify({
          iss: accessKey,
          exp: now + 1800, // 30-minute expiry
          nbf: now - 5,    // valid 5s ago (clock skew tolerance)
        })
      );

      const message = `${header}.${payload}`;
      const signature = crypto
        .createHmac("sha256", secretKey)
        .update(message)
        .digest("base64")
        .replace(/\+/g, "-")
        .replace(/\//g, "_")
        .replace(/=/g, "");

      return `${message}.${signature}`;
    }
    ```

    Critical rules:
    - The JWT signing uses `.digest("base64")` + base64url replacement — NOT `.digest("hex")`. The webhook verification (a different operation) uses hex. Do NOT mix these.
    - KLING_ACCESS_KEY and KLING_SECRET_KEY are two separate env vars — not one key.
    - The function throws (not returns NextResponse) because it is a utility called inside route handlers that handle the error themselves.
    - No TypeScript errors — the file must compile cleanly with `npx tsc --noEmit`.

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Server env → JWT signing | KLING_SECRET_KEY is the signing secret; must never reach client bundle |
| generateKlingToken() → API caller | Returns a JWT string; caller sends it as Bearer token to Kling API |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Information Disclosure | lib/kling-auth.ts | mitigate | File must NEVER be imported from any client component; add `"use server"` guard or ensure it's only imported from `app/api/` routes. The `import crypto from "crypto"` will cause a build error if accidentally bundled for the browser — this is the natural guard. |
| T-02-02 | Tampering | JWT claims (exp/nbf) | mitigate | 30-minute expiry (exp: now+1800) limits the window if a token leaks; nbf=now-5 handles clock skew. Per RESEARCH.md anti-patterns: do NOT cache tokens across requests in a way that allows a leaked token to be reused indefinitely. |
| T-02-03 | Information Disclosure | KLING_SECRET_KEY in error logs | mitigate | The missing-env guard logs "[Kling] KLING_ACCESS_KEY or KLING_SECRET_KEY is not configured" without printing the key value itself. Never log the secretKey value. |
| T-02-04 | Spoofing | Wrong model names in CREDIT_COSTS | accept | If KLING-API-NOTES.md confirms different model name strings, update accordingly; getCreditCost() fallback returns 1 (not an error) for unknown models — acceptable during transition, logged in plan 03 |
